CREATE OR REPLACE PROCEDURE Exp (Var IN emp.first_name%TYPE)
AS
   v_name   emp.first_name%TYPE;
   v_salary emp.salary%TYPE;

BEGIN
   SELECT first_name, salary
   INTO v_name, v_salary
   FROM emp
   WHERE first_name = Var;

   DBMS_OUTPUT.PUT_LINE('Name: ' || v_name);
   DBMS_OUTPUT.PUT_LINE('Salary: ' || v_salary);

EXCEPTION
   WHEN NO_DATA_FOUND THEN
      DBMS_OUTPUT.PUT_LINE('Employee not found');

END Exp;

CREATE OR REPLACE PROCEDURE Sub_Avg (
    cn   IN INT,
    dbms IN INT,
    ifds IN INT
) AS
    total INT;
    per   NUMBER(5,2);
BEGIN
    total := cn + dbms + ifds;
    per := (total / 300) * 100;

    IF per > 75 THEN
        DBMS_OUTPUT.PUT_LINE('DISTINCTION');

    ELSIF per > 65 THEN
        DBMS_OUTPUT.PUT_LINE('FIRST CLASS');

    ELSE
        DBMS_OUTPUT.PUT_LINE('FAIL');
    END IF;

    DBMS_OUTPUT.PUT_LINE('PERCENTAGE IS ' || per);
END;

CREATE OR REPLACE PROCEDURE check_marks (m1 IN NUMBER, m2 IN NUMBER, m3 IN NUMBER)
AS
   Invalid_marks EXCEPTION;
BEGIN
   IF m1 < 0 OR m1 > 100 OR m2 < 0 OR m2 > 100 OR m3 < 0 OR m3 > 100 THEN
      RAISE Invalid_marks;
   ELSIF m1 >= 50 AND m2 >= 50 AND m3 >= 50 THEN
      DBMS_OUTPUT.PUT_LINE('Result: Pass');
   ELSE
      DBMS_OUTPUT.PUT_LINE('Result: Fail');
   END IF;

EXCEPTION
   WHEN Invalid_marks THEN
      DBMS_OUTPUT.PUT_LINE('Enter valid marks in range (0-100)');
END;
/
